function saveQuestionReport(q,type){
  const g=loadGlobal();g.questionReports=Array.isArray(g.questionReports)?g.questionReports:[];
  const item={questionId:q.id||'',type:String(type||'อื่น ๆ'),prompt:q.prompt||'',category:q.category||'',topic:q.topic||q.category||'',position:currentPosition,set:String(currentSet),reportedAt:Date.now()};
  g.questionReports.unshift(item);g.questionReports=g.questionReports.slice(0,100);saveGlobal(g);
  window.KEXAM_ANALYTICS?.track('question_report',{position:currentPosition,set_no:Number.isInteger(Number(currentSet))?Number(currentSet):null,exam_mode:isRandomSet(currentSet)?'random':isWrongSet(currentSet)?'wrong-practice':isWeakSet(currentSet)?'weak-practice':isTopicSet(currentSet)?'topic-practice':'fixed',question_id:item.questionId,report_type:item.type,category:item.category,topic:item.topic,prompt:String(item.prompt).slice(0,240)});
  return item;
}
