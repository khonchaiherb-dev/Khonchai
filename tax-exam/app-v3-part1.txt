function positionProgress(key){
  const old=currentPosition;currentPosition=key;const g=loadGlobal();let completed=0,started=0,best=[];
  for(let i=1;i<=SETS;i++){const s=g['set'+i];if(s){const a=Object.keys(s.answers||{}).length;if(a)started++;if(s.submitted)completed++}const b=bestScore(i);if(b!==null)best.push(b)}
  currentPosition=old;
  return{completed,started,best:best.length?Math.max(...best):null};
}
function weaknessSummary(){
  const attempts=[];
  [...Array(SETS).keys()].map(i=>i+1).concat(RANDOM_SET).forEach(n=>getHistory(n).forEach(x=>{if(x?.category)attempts.push(x)}));
  attempts.sort((a,b)=>(Number(b.submittedAt)||0)-(Number(a.submittedAt)||0));
  const recent=attempts.slice(0,10),m={};
  recent.forEach(x=>Object.entries(x.category||{}).forEach(([cat,v])=>{const total=Number(v?.total)||0,correct=Number(v?.correct)||0;if(!total)return;m[cat]??={total:0,correct:0};m[cat].total+=total;m[cat].correct+=Math.max(0,Math.min(correct,total))}));
  return Object.entries(m).map(([name,v])=>({name,...v,percent:pct(v.correct,v.total)})).sort((a,b)=>a.percent-b.percent||b.total-a.total);
}
function weaknessTopicSummary(){
  const attempts=[];
  [...Array(SETS).keys()].map(i=>i+1).concat(RANDOM_SET).forEach(n=>getHistory(n).forEach(x=>{if(x?.topic)attempts.push(x)}));
  attempts.sort((a,b)=>(Number(b.submittedAt)||0)-(Number(a.submittedAt)||0));
  const recent=attempts.slice(0,10),m={};
  recent.forEach(x=>Object.entries(x.topic||{}).forEach(([topic,v])=>{const total=Number(v?.total)||0,correct=Number(v?.correct)||0;if(!total)return;m[topic]??={total:0,correct:0};m[topic].total+=total;m[topic].correct+=Math.max(0,Math.min(correct,total))}));
  return Object.entries(m).filter(([,v])=>v.total>=2).map(([name,v])=>({name,...v,percent:pct(v.correct,v.total)})).sort((a,b)=>a.percent-b.percent||b.total-a.total||a.name.localeCompare(b.name,'th'));
}
function repeatedMistakeSummary(){
  const stats=loadGlobal().itemStats||{};
  return Object.entries(stats).map(([id,x])=>{
    const attempts=Number(x?.attempts)||0,correct=Number(x?.correct)||0,accuracy=attempts?Math.round(correct*100/attempts):100,wrong=Math.max(0,attempts-correct);
    return{id,attempts,correct,wrong,accuracy,topic:String(x?.topic||x?.category||''),prompt:String(x?.prompt||'')};
  }).filter(x=>x.attempts>=2&&x.wrong>=1).sort((a,b)=>b.wrong-a.wrong||a.accuracy-b.accuracy||b.attempts-a.attempts||a.id.localeCompare(b.id)).slice(0,50);
}
