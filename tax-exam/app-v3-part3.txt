function itemReviewMetric(x){
  const attempts=Number(x?.attempts)||0,correct=Number(x?.correct)||0,answer=Number(x?.answer),choices=Array.isArray(x?.choices)&&x.choices.length===4?x.choices.map(v=>Number(v)||0):[0,0,0,0];
  const accuracy=attempts?Math.round(correct*1000/attempts)/10:0,unused=choices.reduce((n,v,i)=>n+(i!==answer&&v===0?1:0),0;
  let score=0;const flags=[];
  if(attempts>=3&&accuracy<35){score+=32;flags.push('อัตราตอบถูกต่ำ')}
  if(attempts>=5&&accuracy>95){score+=18;flags.push('อัตราตอบถูกสูงมาก')}
  if(unused){score+=Math.min(36,unused*12);flags.push(`ตัวลวงไม่ถูกเลือก ${unused}/3`)}
  score+=Math.min(14,Math.max(0,attempts-2)*2);
  return{accuracy,unused,reviewScore:Math.min(100,Math.round(score)),reviewFlags:flags};
}
function recordItemStats(){
  const g=loadGlobal();g.itemStats=g.itemStats&&typeof g.itemStats==='object'?g.itemStats:{};
  const now=Date.now();
  bank.questions.forEach((q,i)=>{
    if(!q.id)return;const selected=Number(state.answers[i]);
    const x=g.itemStats[q.id]&&typeof g.itemStats[q.id]==='object'?g.itemStats[q.id]:{attempts:0,correct:0,choices:[0,0,0,0]};
    x.attempts=(Number(x.attempts)||0)+1;x.correct=(Number(x.correct)||0)+(selected===q.answer?1:0);
    x.choices=Array.isArray(x.choices)&&x.choices.length===4?x.choices.map(v=>Number(v)||0):[0,0,0,0];
    if(Number.isInteger(selected)&&selected>=0&&selected<4)x.choices[selected]++;
    x.answer=q.answer;x.category=q.category||'';x.topic=q.topic||q.category||'';x.prompt=String(q.prompt||'').slice(0,240);x.lastSeen=now;
    Object.assign(x,itemReviewMetric(x));g.itemStats[q.id]=x;
  });
  saveGlobal(g);
}
