async function loadCompactBank(key=currentPosition){
  const p=POSITIONS[key];if(!p)throw new Error('ยังไม่ได้เลือกตำแหน่งสอบ');
  if(!bankPromises[key])bankPromises[key]=(async()=>{
    const parts=await Promise.all(p.bankFiles.map(async f=>{const r=await fetch(f,{cache:'no-store'});if(!r.ok)throw new Error(`โหลดข้อมูลไม่สำเร็จ: ${f}`);return (await r.text()).trim()}));
    const b64=parts.join('').replace(/\s+/g,'');
    const bin=atob(b64),bytes=new Uint8Array(bin.length);for(let i=0;i<bin.length;i++)bytes[i]=bin.charCodeAt(i);
    if(typeof DecompressionStream==='undefined')throw new Error('เบราว์เซอร์นี้ไม่รองรับการเปิดคลังข้อสอบ กรุณาใช้ Chrome, Edge หรือ Safari รุ่นปัจจุบัน');
    const ds=new DecompressionStream('gzip');const data=await new Response(new Blob([bytes]).stream().pipeThrough(ds)).json();
    if(!Array.isArray(data.q)||data.q.length!==10||data.q.some(x=>!Array.isArray(x)||x.length!==100))throw new Error('คลังข้อสอบไม่ครบ 10 ชุด ชุดละ 100 ข้อ');return data
  })().catch(e=>{delete bankPromises[key];throw e});
  return bankPromises[key];
}
function questionId(si,qi){const prefix=currentPosition==='revenue-academic'?'RA':'TA';return `${prefix}-S${String(si+1).padStart(2,'0')}-Q${String(qi+1).padStart(3,'0')}`}
function parseQuestionId(id){const m=/^(RA|TA)-S(\d{2})-Q(\d{3})$/.exec(String(id||''));if(!m)return null;const expected=currentPosition==='revenue-academic'?'RA':'TA';if(m[1]!==expected)return null;const si=Number(m[2])-1,qi=Number(m[3])-1;return si>=0&&si<10&&qi>=0&&qi<100?{si,qi}:null}
function decodeQuestion(c,r,si,qi){const t=c.s;return{id:questionId(si,qi),number:si*100+qi+1,sourceSet:si+1,sourceIndex:qi+1,category:t[r[0]],prompt:t[r[1]],choices:r[2].map(x=>t[x]),answer:r[3],explanation:t[r[4]],wrongReasons:t[r[5]]}}
async function loadSet(n){const c=await loadCompactBank(),rows=c.q[n-1];return{set:n,title:`ชุดที่ ${n}`,questions:rows.map((r,i)=>decodeQuestion(c,r,n-1,i))}}
function randomize(arr){const a=[...arr];for(let i=a.length-1;i>0;i--){let j;if(globalThis.crypto?.getRandomValues){const x=new Uint32Array(1);crypto.getRandomValues(x);j=x[0]%(i+1)}else j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]]}return a}
function buildRandomPlan(c){
  const pools={};c.q.forEach((set,si)=>set.forEach((r,qi)=>{const cat=String(c.s[r[0]]||'ไม่ระบุ');(pools[cat]??=[]).push(`${si}:${qi}`)}));
  const entries=Object.entries(pools).map(([cat,ids])=>({cat,ids,raw:ids.length/10,quota:Math.floor(ids.length/10)}));
  let left=100-entries.reduce((n,x)=>n+x.quota,0);entries.sort((a,b)=>(b.raw-b.quota)-(a.raw-a.quota)||b.ids.length-a.ids.length);
  for(let i=0;i<entries.length&&left>0;i++,left--)entries[i].quota++;
  const chosen=[];entries.forEach(x=>chosen.push(...randomize(x.ids).slice(0,x.quota)));
  if(chosen.length!==100)throw new Error('ไม่สามารถจัดสัดส่วนข้อสอบสุ่มให้ครบ 100 ข้อได้');
  return randomize(chosen);
}
async function loadRandomSet(savedIds=null){
  const c=await loadCompactBank();let ids=Array.isArray(savedIds)?savedIds.map(String):null;
  const valid=id=>{const m=/^(\d+):(\d+)$/.exec(id);if(!m)return false;const si=+m[1],qi=+m[2];return !!c.q?.[si]?.[qi]};
  if(!ids||ids.length!==100||new Set(ids).size!==100||ids.some(id=>!valid(id)))ids=buildRandomPlan(c);
  const questions=ids.map(id=>{const [si,qi]=id.split(':').map(Number);return decodeQuestion(c,c.q[si][qi],si,qi)});
  return{set:RANDOM_SET,title:'สุ่มจำลองสอบ 100 ข้อ',questions,randomIds:ids};
}

function questionFromId(c,id){const p=parseQuestionId(id);return p&&c.q?.[p.si]?.[p.qi]?decodeQuestion(c,c.q[p.si][p.qi],p.si,p.qi):null}
function collectHistoryWrongIds(){
  const out=[];
  [...Array(SETS).keys()].map(i=>i+1).concat([RANDOM_SET,WRONG_SET,WEAK_SET]).forEach(n=>getHistory(n).forEach(h=>{if(Array.isArray(h?.wrongIds))out.push(...h.wrongIds)}));
  return out;
}
async function collectWrongIds(c){
  const ids=new Set(collectHistoryWrongIds().filter(id=>questionFromId(c,id)));
  for(let sn=1;sn<=SETS;sn++){
    const st=getSetState(sn);if(!st?.submitted)continue;
    const set=c.q[sn-1]||[];
    set.forEach((r,qi)=>{if(st.answers?.[qi]!==r[3])ids.add(questionId(sn-1,qi))});
  }
  const rs=getSetState(RANDOM_SET);
  if(rs?.submitted&&Array.isArray(rs.randomIds))rs.randomIds.forEach((raw,i)=>{const m=/^(\d+):(\d+)$/.exec(String(raw));if(!m)return;const si=+m[1],qi=+m[2],r=c.q?.[si]?.[qi];if(r&&rs.answers?.[i]!==r[3])ids.add(questionId(si,qi))});
  for(const kind of [WRONG_SET,WEAK_SET]){
    const st=getSetState(kind);if(!st?.submitted||!Array.isArray(st.practiceIds))continue;
    st.practiceIds.forEach((id,i)=>{const q=questionFromId(c,id);if(q&&st.answers?.[i]!==q.answer)ids.add(q.id)});
  }
  return [...ids];
}
function buildWeakPlan(c,category,limit=50){
  const ids=[];c.q.forEach((set,si)=>set.forEach((r,qi)=>{if(String(c.s[r[0]]||'')===String(category))ids.push(questionId(si,qi))}));
  if(!ids.length)throw new Error('ไม่พบข้อสอบในหมวดที่เลือก');
  return randomize(ids).slice(0,Math.min(limit,ids.length));
}
async function loadPracticeSet(kind,{category=null,savedIds=null}={}){
  const c=await loadCompactBank();let ids=Array.isArray(savedIds)?savedIds.map(String).filter(id=>questionFromId(c,id)):null;
  if(kind===WRONG_SET){
    if(!ids?.length){ids=await collectWrongIds(c);ids=randomize(ids).slice(0,Math.min(100,ids.length))}
    if(!ids.length)throw new Error('ยังไม่พบข้อที่เคยตอบผิดในอุปกรณ์นี้ กรุณาทำและส่งข้อสอบอย่างน้อย 1 ชุดก่อน');
    return{set:WRONG_SET,title:`ฝึกเฉพาะข้อที่เคยผิด ${ids.length} ข้อ`,questions:ids.map(id=>questionFromId(c,id)),practiceIds:ids};
  }
  if(kind===WEAK_SET){
    if(!category)throw new Error('ยังไม่มีข้อมูลหมวดที่ควรทบทวน');
    if(!ids?.length)ids=buildWeakPlan(c,category,50);
    return{set:WEAK_SET,title:`ฝึกหมวดที่ควรทบทวน · ${category}`,questions:ids.map(id=>questionFromId(c,id)),practiceIds:ids,practiceCategory:category};
  }
  throw new Error('ไม่รู้จักโหมดฝึกนี้');
}
