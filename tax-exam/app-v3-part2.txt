async function loadPublicationManifest(){
  if(publicationManifest)return publicationManifest;
  if(!publicationManifestPromise)publicationManifestPromise=(async()=>{
    const r=await fetch('question-publication-manifest.json',{cache:'no-store'});
    if(!r.ok)throw new Error('โหลด Publication Manifest ไม่สำเร็จ');
    const m=await r.json();
    if(Number(m?.schema_version)!==1||!m?.positions)throw new Error('Publication Manifest ไม่ถูกต้อง');
    for(const key of Object.keys(POSITIONS)){
      const ids=m.positions?.[key]?.published_ids;
      if(!Array.isArray(ids))throw new Error(`Publication Manifest ไม่มีรายการ Published ของ ${key}`);
      publishedIdSets[key]=new Set(ids.map(String));
    }
    publicationManifest=m;return m;
  })().catch(e=>{publicationManifestPromise=null;throw e});
  return publicationManifestPromise;
}

function randomize(arr){const a=[...arr];for(let i=a.length-1;i>0;i--){let j;if(globalThis.crypto?.getRandomValues){const x=new Uint32Array(1);crypto.getRandomValues(x);j=x[0]%(i+1)}else j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]]}return a}
function allocateSample(groups,target){
  const total=Object.values(groups).reduce((n,ids)=>n+ids.length,0);if(!total||target<=0)return[];
  const entries=Object.entries(groups).map(([name,ids])=>{const raw=ids.length/total*target;return{name,ids,raw,quota:Math.floor(raw)}});
  let left=target-entries.reduce((n,x)=>n+x.quota,0);
  entries.sort((a,b)=>(b.raw-b.quota)-(a.raw-a.quota)||b.ids.length-a.ids.length||a.name.localeCompare(b.name,'th'));
  for(let i=0;left>0&&entries.length;i=(i+1)%entries.length,left--)if(entries[i].quota<entries[i].ids.length)entries[i].quota++;
  const chosen=[];entries.forEach(x=>chosen.push(...randomize(x.ids).slice(0,x.quota)));return chosen;
}
function buildRandomPlan(c){
  const cats={};let total=0;
  c.q.forEach((set,si)=>set.forEach((r,qi)=>{const id=questionId(si,qi);if(!isPublishedQuestionId(id))return;const cat=String(c.s[r[0]]||'ไม่ระบุ'),prompt=String(c.s[r[1]]||''),topic=deriveTopic(cat,prompt);cats[cat]??={};(cats[cat][topic]??=[]).push(`${si}:${qi}`);total++}));
  const catEntries=Object.entries(cats).map(([cat,topics])=>{const size=Object.values(topics).reduce((n,a)=>n+a.length,0),raw=size/total*100;return{cat,topics,size,raw,quota:Math.floor(raw)}});
  let left=100-catEntries.reduce((n,x)=>n+x.quota,0);
  catEntries.sort((a,b)=>(b.raw-b.quota)-(a.raw-a.quota)||b.size-a.size||a.cat.localeCompare(b.cat,'th'));
  for(let i=0;left>0&&catEntries.length;i=(i+1)%catEntries.length,left--)if(catEntries[i].quota<catEntries[i].size)catEntries[i].quota++;
  const chosen=[];catEntries.forEach(x=>chosen.push(...allocateSample(x.topics,x.quota)));
  if(chosen.length!==100||new Set(chosen).size!==100)throw new Error('ไม่สามารถจัดสัดส่วนข้อสอบสุ่มตามหมวดและหัวข้อย่อยให้ครบ 100 ข้อได้');
  return randomize(chosen);
}

  for(let sn=1;sn<=SETS;sn++){
    const st=getSetState(sn);if(!st?.submitted)continue;
    const set=c.q[sn-1]||[];
    set.forEach((r,qi)=>{if(st.answers?.[qi]!==r[3])ids.add(questionId(sn-1,qi))});
  }
  const rs=getSetState(RANDOM_SET);
  if(rs?.submitted&&Array.isArray(rs.randomIds))rs.randomIds.forEach((raw,i)=>{const m=/^(\d+):(\d+)$/.exec(String(raw));if(!m)return;const si=+m[1],qi=+m[2],r=c.q?.[si]?.[qi];if(r&&rs.answers?.[i]!==r[3])ids.add(questionId(si,qi))});
  for(const kind of [WRONG_SET,WEAK_SET,TOPIC_SET]){
    const st=getSetState(kind);if(!st?.submitted||!Array.isArray(st.practiceIds))continue;

function buildWeakPlan(c,category,limit=50){
  const ids=[];c.q.forEach((set,si)=>set.forEach((r,qi)=>{const id=questionId(si,qi);if(isPublishedQuestionId(id)&&String(c.s[r[0]]||'')===String(category))ids.push(id)}));

}
function buildTopicPlan(c,topic,limit=40){

async function loadPracticeSet(kind,{category=null,topic=null,savedIds=null}={}){
  const c=await loadCompactBank();let ids=Array.isArray(savedIds)?savedIds.map(String).filter(id=>questionFromId(c,id)):null;
  if(kind===WRONG_SET){
    if(!ids?.length){ids=await collectWrongIds(c);ids=randomize(ids).slice(0,Math.min(100,ids.length))}
    if(!ids.length)throw new Error('ยังไม่พบข้อที่เคยตอบผิดในอุปกรณ์นี้ กรุณาทำและส่งข้อสอบอย่างน้อย 1 ชุดก่อน');
    return{set:WRONG_SET,title:`ฝึกเฉพาะข้อที่เคยผิด ${ids.length} ข้อ`,questions:ids.map(id=>questionFromId(c,id)),practiceIds:ids};
  }
  if(kind===WEAK_SET){
    if(!category)throw new Error('ยังไม่มีข้อมูลหมวดที่ควรทบทวน');
    if(!ids?.length)ids=buildWeakPlan(c,category,50);
    return{set:WEAK_SET,title:`ฝึกหมวดที่ควรทบทวน · ${category}`,questions:ids.map(id=>questionFromId(c,id)),practiceIds:ids,practiceCategory:category};
  }
  if(kind===TOPIC_SET){
    if(!topic)throw new Error('ยังไม่มีข้อมูลหัวข้อย่อยที่ควรทบทวน');
    if(!ids?.length)ids=buildTopicPlan(c,topic,40);

  }
  throw new Error('ไม่รู้จักโหมดฝึกนี้');
}
