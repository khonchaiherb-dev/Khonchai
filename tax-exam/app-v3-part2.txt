async function loadCompactBank(key=currentPosition){
  const p=POSITIONS[key];if(!p)throw new Error('ยังไม่ได้เลือกตำแหน่งสอบ');
  if(!bankPromises[key])bankPromises[key]=(async()=>{
    const parts=await Promise.all(p.bankFiles.map(async f=>{const r=await fetch(f,{cache:'no-store'});if(!r.ok)throw new Error(`โหลดข้อมูลไม่สำเร็จ: ${f}`);return (await r.text()).trim()}));
    const b64=parts.join('').replace(/\s+/g,'');
    const bin=atob(b64),bytes=new Uint8Array(bin.length);for(let i=0;i<bin.length;i++)bytes[i]=bin.charCodeAt(i);
    if(typeof DecompressionStream==='undefined')throw new Error('เบราว์เซอร์นี้ไม่รองรับการเปิดคลังข้อสอบ กรุณาใช้ Chrome, Edge หรือ Safari รุ่นปัจจุบัน');
    const ds=new DecompressionStream('gzip');const data=await new Response(new Blob([bytes]).stream().pipeThrough(ds)).json();
    if(!Array.isArray(data.q)||data.q.length!==10||data.q.some(x=>!Array.isArray(x)||x.length!==100))throw new Error('คลังข้อสอบไม่ครบ 10 ชุด ชุดละ 100 ข้อ');return data
  })().catch(e=>{delete bankPromises[key];throw e});
  return bankPromises[key];
}
function decodeQuestion(c,r,si,qi){const t=c.s;return{number:si*100+qi+1,sourceSet:si+1,sourceIndex:qi+1,category:t[r[0]],prompt:t[r[1]],choices:r[2].map(x=>t[x]),answer:r[3],explanation:t[r[4]],wrongReasons:t[r[5]]}}
async function loadSet(n){const c=await loadCompactBank(),rows=c.q[n-1];return{set:n,title:`ชุดที่ ${n}`,questions:rows.map((r,i)=>decodeQuestion(c,r,n-1,i))}}
function randomize(arr){const a=[...arr];for(let i=a.length-1;i>0;i--){let j;if(globalThis.crypto?.getRandomValues){const x=new Uint32Array(1);crypto.getRandomValues(x);j=x[0]%(i+1)}else j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]]}return a}
function buildRandomPlan(c){
  const pools={};c.q.forEach((set,si)=>set.forEach((r,qi)=>{const cat=String(c.s[r[0]]||'ไม่ระบุ');(pools[cat]??=[]).push(`${si}:${qi}`)}));
  const entries=Object.entries(pools).map(([cat,ids])=>({cat,ids,raw:ids.length/10,quota:Math.floor(ids.length/10)}));
  let left=100-entries.reduce((n,x)=>n+x.quota,0);entries.sort((a,b)=>(b.raw-b.quota)-(a.raw-a.quota)||b.ids.length-a.ids.length);
  for(let i=0;i<entries.length&&left>0;i++,left--)entries[i].quota++;
  const chosen=[];entries.forEach(x=>chosen.push(...randomize(x.ids).slice(0,x.quota)));
  if(chosen.length!==100)throw new Error('ไม่สามารถจัดสัดส่วนข้อสอบสุ่มให้ครบ 100 ข้อได้');
  return randomize(chosen);
}
async function loadRandomSet(savedIds=null){
  const c=await loadCompactBank();let ids=Array.isArray(savedIds)?savedIds.map(String):null;
  const valid=id=>{const m=/^(\d+):(\d+)$/.exec(id);if(!m)return false;const si=+m[1],qi=+m[2];return !!c.q?.[si]?.[qi]};
  if(!ids||ids.length!==100||new Set(ids).size!==100||ids.some(id=>!valid(id)))ids=buildRandomPlan(c);
  const questions=ids.map(id=>{const [si,qi]=id.split(':').map(Number);return decodeQuestion(c,c.q[si][qi],si,qi)});
  return{set:RANDOM_SET,title:'สุ่มจำลองสอบ 100 ข้อ',questions,randomIds:ids};
}
